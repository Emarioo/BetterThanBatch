```rust
import "File" as IO // file system, OS
import "System" as SYS // hardware, os

// System.btb



// compile time constants because IO.platform is a pure function (no side effects, same input same output, it's a compiler function)

import qq

@scope
@import(LIB) global WINDOWS := SYS.platform() == "Windows"
LINUX := SYS.platform() == "Linux"



// main.btb



// compile time if expression
if WINDOWS
    #import "TracyWin32"
else if LINUX
    #import "TracyLinux"

InitTracy = TracyWin32 or TracyLinux

InitTracy(variable)

// File.btb
if WINDOWS
    #import "Windows"
else if LINUX
    #import "Linux"

ReadFile : fn () = {
    if WINDOWS {
        WinReadFile(...)
    } else if LINUX {
        LinuxReadFile(...)
    }
}
```
